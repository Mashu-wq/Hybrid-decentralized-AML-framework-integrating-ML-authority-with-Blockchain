  ---
  Why fraud scores are low on first transaction

  The transaction-service enriches raw Kafka messages with Redis data (velocity, prior transaction count, KYC risk score). For a new
  customer with no transaction history, those Redis fields are zero. The model then sees sparse features and outputs a low score
  (real LightGBM output: 0.006). This is correct behavior.

  To trigger alerts, you have two options:
  - Option A: Build up transaction history first (send 3–5 normal transactions), then send suspicious ones
  - Option B: Update the customer to CRITICAL risk (threshold 0.25) — even a moderate score can trigger an alert

================================================================================
KNOWN LIMITATIONS FOR TESTING
================================================================================

1. TRANSACTIONS via Kafka only: The transaction service has no REST HTTP endpoint
   for ingesting transactions. All transactions must be published to the Kafka
   topic "transactions.raw". Use the Kafka CLI or a Kafka client library.

2. ML service warmup: The first prediction after startup may be slower (~2-3s)
   as models load into memory. Subsequent calls are fast (<100ms).

3. Blockchain latency: Fabric block commit takes ~400-900ms. After submitting
   a transaction to Kafka, wait 2-3 seconds before checking the audit trail.

4. KYC risk propagation: After updating a customer's KYC risk level, there is
   a ~5-second Redis TTL before the new threshold takes effect for the next
   transaction. This is the cache TTL, not a bug.

5. SAR generation requires S3: The SAR PDF upload step (Phase 5.8) requires
   a configured S3 bucket. If S3 is not configured, the blockchain hash anchor
   step will succeed but the PDF storage will fail.

  import pandas as pd
  import numpy as np
  import warnings 
  warnings.filterwarnings('ignore')

  CSV_PATH = "/content/drive/MyDrive/AML_thesis/PS_20174392719_1491204439457_log.csv"
  
  df = pd.read_csv(CSV_PATH)
  print(f"Shape: {df.shape}")

  print(f"\nFraud by type:\n{df.groupby('type')['isFraud'].sum()}")

  Cell 3 — Feature engineering

  # Keep only fraud-relevant transaction types
  df = df[df["type"].isin(["TRANSFER", "CASH_OUT"])].copy()
  print(f"After filter: {len(df):,} rows")
  print(f"Fraud count: {df['isFraud'].sum():,}")

  # Engineer features
  df["type_is_transfer"]    = (df["type"] == "TRANSFER").astype(int)
  df["type_is_cashout"]     = (df["type"] == "CASH_OUT").astype(int)
  df["hour_of_day"]         = df["step"] % 24

  # Balance anomaly — KEY features for fraud detection
  df["error_balance_orig"]  = df["oldbalanceOrg"] - df["amount"] - df["newbalanceOrig"]
  df["error_balance_dest"]  = df["oldbalanceDest"] + df["amount"] - df["newbalanceDest"]
  df["orig_balance_ratio"]  = df["amount"] / (df["oldbalanceOrg"] + 1)
  df["dest_balance_ratio"]  = df["amount"] / (df["oldbalanceDest"] + 1)
  df["is_zero_orig_after"]  = (df["newbalanceOrig"] == 0).astype(int)

  FEATURES = [
      "amount",
      "oldbalanceOrg", "newbalanceOrig",
      "oldbalanceDest", "newbalanceDest",
      "type_is_transfer", "type_is_cashout",
      "hour_of_day",
      "error_balance_orig",  "error_balance_dest",
      "orig_balance_ratio",  "dest_balance_ratio",
      "is_zero_orig_after",  "is_zero_dest_before",
  ]   

  X = df[FEATURES].values
  y = df["isFraud"].values
  print(f"\nFeature matrix: {X.shape}")
  print(f"Features: {FEATURES}")

  Cell 4 — Train/test split + SMOTE
  from imblearn.over_sampling import SMOTE

  # Temporal split (80/20) — no random shuffle to avoid data leakage
  split = int(len(X) * 0.8)
  X_train, X_test = X[:split], X[split:]
  y_train, y_test = y[:split], y[split:]
  
  print(f"Train: {len(X_train):,}  |  Test: {len(X_test):,}")
  print(f"Train fraud rate: {y_train.mean():.4f}")

  # SMOTE for class imbalance
  print("\nApplying SMOTE (takes 2-3 minutes)...")
  sm = SMOTE(random_state=42, n_jobs=-1)
  X_train_res, y_train_res = sm.fit_resample(X_train, y_train)
  print(f"After SMOTE: {len(X_train_res):,} samples") 
  print(f"Fraud balance: {y_train_res.mean():.2f}")

  Cell 5 — Train all 3 models

  import time
  from lightgbm import LGBMClassifier
  from xgboost import XGBClassifier
  from sklearn.ensemble import RandomForestClassifier
  from sklearn.metrics import (precision_score, recall_score, f1_score,
                               roc_auc_score, average_precision_score)

  def evaluate(name, model, X_test, y_test):
      proba = model.predict_proba(X_test)[:, 1]
      pred  = (proba >= 0.5).astype(int)
      print(f"\n{'='*50}")
      print(f"  {name}")
      print(f"{'='*50}")
      print(f"  Precision : {precision_score(y_test, pred)*100:.2f}%")
      print(f"  Recall    : {recall_score(y_test, pred)*100:.2f}%")
      print(f"  F1        : {f1_score(y_test, pred)*100:.2f}%")
      print(f"  ROC-AUC   : {roc_auc_score(y_test, proba)*100:.2f}%")
      print(f"  PR-AUC    : {average_precision_score(y_test, proba)*100:.2f}%")
      return proba

  # LightGBM
  print("Training LightGBM...")
  t0 = time.time()
  lgbm = LGBMClassifier(n_estimators=1000, num_leaves=1000,
                         learning_rate=0.05, is_unbalance=False,
                         random_state=42, n_jobs=-1, verbose=-1)
  lgbm.fit(X_train_res, y_train_res)
  print(f"Done in {time.time()-t0:.0f}s")
  lgbm_proba = evaluate("LightGBM (PaySim)", lgbm, X_test, y_test)

  # Random Forest
  print("\nTraining Random Forest...")
  rf = RandomForestClassifier(n_estimators=500, max_depth=12,
                               class_weight="balanced",
                               random_state=42, n_jobs=-1)
  rf.fit(X_train_res, y_train_res)
  print(f"Done in {time.time()-t0:.0f}s")
  rf_proba = evaluate("Random Forest (PaySim)", rf, X_test, y_test)
  
  # XGBoost
  print("\nTraining XGBoost...")
  t0 = time.time()
  xgb = XGBClassifier(n_estimators=500, learning_rate=0.05,
                       max_depth=6, random_state=42,
                       n_jobs=-1, eval_metric="logloss",
                       verbosity=0)
  xgb.fit(X_train_res, y_train_res)
  print(f"Done in {time.time()-t0:.0f}s")
